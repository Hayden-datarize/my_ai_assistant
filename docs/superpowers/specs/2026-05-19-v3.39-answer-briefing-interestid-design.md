```ts
/**
 * v3.39 T1: schema v8 → v9 — Answer/Briefing interestId 도입 (User storage는 schemaVersion만 lift).
 * - Answer/Briefing은 별도 storage (loadAnswers/loadBriefings boundary에서 normalize) → User 객체 자체엔 ripple 0.
 * - chain superset (v3.25 T2 패턴): V2~V8 모든 early-return이 v9 user에 same-reference pass-through.
 * - [v2 — Codex P0-1]: V2~V8 함수 안에 `r.schemaVersion === 9 → return r as User` early-return 추가 (downgrade 차단).
 */
export function migrateUserToV9(u: unknown): User {
  const r = u as Record<string, unknown> & { schemaVersion?: number };

  if (r.schemaVersion === 9) {
    return r as unknown as User; // idempotent
  }

  // v8까지 lift
  const v8 = migrateUserToV8(u);
  return { ...(v8 as unknown as Record<string, unknown>), schemaVersion: 9 } as unknown as User;
}
```

**[v2 — Codex P0-3 흡수] `matchesInterest(text, id)` 신규 helper 필수**

spec v1 `matchKeyword(text, interestId)` 가정은 **잘못**. 현재 `src/utils/interestKeywords.ts`의 `matchKeyword`은 keyword token (e.g., '채용', 'HR')용이고 interest id (`ai_ml`, `leadership`, `hr_system`)를 직접 검색 텍스트에 매칭하면 대부분 false negative.

```ts
// src/utils/interestKeywords.ts 신규
/**
 * v3.39: text가 interestId 분야에 속하는지 판정.
 * - id를 INTERESTS.find로 validate → invalid면 false
 * - id의 `interestKeywords(id)` 토큰 목록 중 하나라도 text에 match → true
 * - text/keyword 모두 lowercase 정규화
 */
export function matchesInterest(text: string, id: string): boolean {
  if (validateInterestId(id) === 'unknown') return false;
  const keywords = interestKeywords(id);
  const hay = text.toLowerCase();
  return keywords.some(kw => matchKeyword(hay, kw));
}
```

`matchKeyword(text, id)` 직접 호출은 spec v2에서 **모두 `matchesInterest(text, id)`로 교체**: §3.1.2 chat-summary / §3.2 Briefing storage 둘 다.
