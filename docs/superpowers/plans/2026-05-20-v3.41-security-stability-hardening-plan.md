```ts
export async function getAppCheckToken(): Promise<string | null> {
  if (!appCheck) return null;
  try {
    const result = await getToken(appCheck, /* forceRefresh */ false);
    return result.token;
  } catch (err) {
    console.warn('[appCheck] getToken failed', err);
    return null;
  }
}
```

```ts
export function parseEvaluationResponse(text: string): { score: 1 | 2 | 3 | 4 | 5; feedback: string } {
  const raw = parseJsonText<unknown>(text);
  let score: 1 | 2 | 3 | 4 | 5 = 3;
  if (raw && typeof raw === 'object' && 'score' in raw) {
    const s = Number((raw as { score: unknown }).score);
    if (Number.isInteger(s) && s >= 1 && s <= 5) score = s as 1 | 2 | 3 | 4 | 5;
    else console.warn('[parseEvaluationResponse] invalid score', s);
  }
  let feedback = '';
  if (raw && typeof raw === 'object' && 'feedback' in raw) {
    const f = (raw as { feedback: unknown }).feedback;
    if (typeof f === 'string') feedback = f.slice(0, 500);
  }
  return { score, feedback };
}
```

```ts
const SAFE_SCHEMES = ['http:', 'https:'];

export function isSafeUrl(value: unknown): value is string {
  if (typeof value !== 'string') return false;
  try {
    const u = new URL(value);
    return SAFE_SCHEMES.includes(u.protocol);
  } catch {
    return false;
  }
}
```
