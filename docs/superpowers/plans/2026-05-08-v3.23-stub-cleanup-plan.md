```ts
/**
 * v3.23 T1: schema v5 → v6 — `User.insights[]` 필드 추가.
 * - insights: Insight[] 신규 필드 (default []).
 * - idempotent: 이미 v6면 그대로 반환 (early return, same reference).
 * - 손상된 insights (non-array / NaN) → default [] 복구.
 */
export function migrateUserToV6(u: unknown): User {
  const v = u as Omit<Partial<User>, 'schemaVersion'> & { schemaVersion?: number; insights?: unknown };
  if (v?.schemaVersion === 6) return u as User;
  // v3/v4/v5 chain — V5까지 끌어올림
  const v5 = migrateUserToV5(u);
  // v5 → v6: insights ??= [], 손상 시 default
  const insights = Array.isArray((v5 as { insights?: unknown }).insights)
    ? (v5 as { insights: unknown[] }).insights
    : [];
  return {
    ...v5,
    schemaVersion: 6 as const,
    insights: insights as Insight[],
  } as User;
}
```

```ts
const USAGE_KEY = 'dg_gemini_usage';
const CAP_KEY = 'dg_gemini_cap';
const DEFAULT_CAP = 50;
const CAP_MIN = 10;
const CAP_MAX = 200;

interface Usage { date: string; count: number; }

function clampCap(n: number): number {
  if (Number.isNaN(n)) return DEFAULT_CAP;
  return Math.max(CAP_MIN, Math.min(CAP_MAX, Math.round(n)));
}

export function getGeminiCap(): number {
  const raw = localStorage.getItem(CAP_KEY);
  if (raw === null) return DEFAULT_CAP;
  return clampCap(Number(raw));
}

export function setGeminiCap(n: number): void {
  localStorage.setItem(CAP_KEY, String(clampCap(n)));
}

function load(): Usage {
  const today = getKstDateStr();
  try {
    const raw = localStorage.getItem(USAGE_KEY);
    if (!raw) return { date: today, count: 0 };
    const parsed = JSON.parse(raw) as Usage;
    if (parsed.date !== today) return { date: today, count: 0 };
    if (typeof parsed.count !== 'number' || !Number.isFinite(parsed.count) || parsed.count < 0) {
      return { date: today, count: 0 };
    }
    return { date: parsed.date, count: Math.floor(parsed.count) };
  } catch {
    return { date: today, count: 0 };
  }
}

  localStorage.setItem(USAGE_KEY, JSON.stringify(u));

  return load().count;
```

```ts
const KEY_PREFIX = 'dg.statsCache.';

export interface StatsFingerprint {
  totalAnswers: number;
  longestStreak: number;
  // codex P1-4 fix: byInterest top key + count 추가 → narrative-relevant 변경 (분야 분포 변동) catch
  byInterestTopKey: string;   // ''(empty) if no interests

export interface StatsCacheEntry {
  range: 7 | 30;
  fingerprint: StatsFingerprint;
  highlight?: string;
  narrative?: string;
  cachedAt: string;
}

function key(range: 7 | 30): string {
  return `${KEY_PREFIX}${getKstDateStr()}.${range}`;
}

export function getStatsCache(range: 7 | 30, fp: StatsFingerprint): StatsCacheEntry | null {
  try {
    const raw = localStorage.getItem(key(range));
    if (!raw) return null;
    const parsed = JSON.parse(raw) as StatsCacheEntry;
    if (parsed.range !== range) return null;
    // codex P1-4 fix: byInterest top key + count 추가 비교 (narrative-relevant 변동 catch)
    if (parsed.fingerprint.totalAnswers !== fp.totalAnswers ||
        parsed.fingerprint.longestStreak !== fp.longestStreak ||
        parsed.fingerprint.byInterestTopKey !== fp.byInterestTopKey ||
        parsed.fingerprint.byInterestTopCount !== fp.byInterestTopCount) return null;
    return parsed;
  } catch {
    return null;
  }
}

export function setStatsCache(
  range: 7 | 30,
  fp: StatsFingerprint,
  data: { highlight?: string; narrative?: string }
): void {
  // cleanup: 다른 KST date prefix 키 일괄 삭제 (오늘 entry만 보존)
  const todayKey = key(range);
  for (let i = localStorage.length - 1; i >= 0; i--) {
    const k = localStorage.key(i);
    if (k && k.startsWith(KEY_PREFIX) && k !== todayKey && !k.startsWith(KEY_PREFIX + getKstDateStr())) {
      localStorage.removeItem(k);
    }
  }
  const entry: StatsCacheEntry = {
    range, fingerprint: fp, ...data, cachedAt: new Date().toISOString(),
  };
  localStorage.setItem(todayKey, JSON.stringify(entry));
}

export function clearStatsCache(): void {
  for (let i = localStorage.length - 1; i >= 0; i--) {
    const k = localStorage.key(i);
    if (k?.startsWith(KEY_PREFIX)) localStorage.removeItem(k);
  }
}
```

```ts
// src/utils/statsAggregate.ts
import { loadAnswers } from '../state/persistence';
import { getKstDateStr } from './dates';

export interface StatsRange {
  totalAnswers: number;
  longestStreak: number;
  activeInterests: number;
  avgPerDay: number;
  byInterest: Array<{ id: string; count: number }>;
  daily?: number[];  // 30만
}

function kstDateOf(iso: string): string {
  // createdAt(UTC ISO) → KST date string YYYY-MM-DD
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return '';
  return new Intl.DateTimeFormat('en-CA', { timeZone: 'Asia/Seoul', year: 'numeric', month: '2-digit', day: '2-digit' }).format(d);
}

// codex P0-2 fix: KST anchor — `T00:00:00` UTC parse가 KST에서 off-by-one 발생.
// `+09:00` 명시 또는 KST 정오(`T12:00:00+09:00`)를 anchor로 두고 day 산술.
function dateNDaysAgo(today: string, n: number): string {
  // today = KST date 'YYYY-MM-DD'. anchor를 KST 정오로 두면 DST 없는 KST에서 day 산술 안전.
  const anchor = new Date(`${today}T12:00:00+09:00`);
  anchor.setDate(anchor.getDate() - n);
  // 다시 KST date로 변환
  return new Intl.DateTimeFormat('en-CA', { timeZone: 'Asia/Seoul', year: 'numeric', month: '2-digit', day: '2-digit' }).format(anchor);
}

export function getStatsRange(days: 7 | 30): StatsRange {
  const today = getKstDateStr();
  const fromDate = dateNDaysAgo(today, days - 1);
  const all = loadAnswers();

  // KST date 기준 filter
  const inRange = all.filter(a => {
    const d = kstDateOf(a.createdAt);
    return d >= fromDate && d <= today;
  });

  const totalAnswers = inRange.length;
  const avgPerDay = Math.round((totalAnswers / days) * 10) / 10;

  // 분야별 카운트 (Answer.type 또는 questionId 매핑)
  const byInterestMap = new Map<string, number>();
  for (const a of inRange) {
    const id = a.type ?? 'unknown';
    byInterestMap.set(id, (byInterestMap.get(id) ?? 0) + 1);
  }
  const byInterest = Array.from(byInterestMap.entries())
    .map(([id, count]) => ({ id, count }))
    .sort((a, b) => b.count - a.count)
    .slice(0, 5);
  const activeInterests = byInterestMap.size;

  // longestStreak — 일별 답변 있는 날 streak (range 안에서)
  const dateSet = new Set(inRange.map(a => kstDateOf(a.createdAt)));
  let longestStreak = 0;
  let curStreak = 0;
  for (let i = days - 1; i >= 0; i--) {
    const d = dateNDaysAgo(today, i);
    if (dateSet.has(d)) {
      curStreak += 1;
      longestStreak = Math.max(longestStreak, curStreak);
    } else {
      curStreak = 0;
    }
  }

  // daily — range=30만
  const daily = days === 30
    ? Array.from({ length: 30 }, (_, i) => {
        const d = dateNDaysAgo(today, 29 - i);
        return inRange.filter(a => kstDateOf(a.createdAt) === d).length;
      })
    : undefined;

  return { totalAnswers, longestStreak, activeInterests, avgPerDay, byInterest, daily };
}
```

```ts
import { escapeHtml } from '../utils/escapeHtml';

export interface ChatPreviewBubbleOpts {

  secondaryLabel: string;
  onPrimary: () => void;
  onSecondary: () => void;
}

export function renderChatPreviewBubble(opts: ChatPreviewBubbleOpts): HTMLDivElement {
  const container = document.getElementById('chatMessages');
  if (!container) throw new Error('chatMessages container not found');

  const bubble = document.createElement('div');
  bubble.className = 'chat-message chat-message-ai chat-preview-bubble';
  // eslint-disable-next-line no-restricted-syntax -- escapeHtml applied
  bubble.innerHTML = `
    <div class="chat-preview-text">${escapeHtml(opts.text)}</div>
    <div class="chat-preview-actions">
      <button type="button" class="btn btn-primary preview-primary">${escapeHtml(opts.primaryLabel)}</button>
      <button type="button" class="btn btn-secondary preview-secondary">${escapeHtml(opts.secondaryLabel)}</button>
    </div>
  `;
  bubble.querySelector('.preview-primary')?.addEventListener('click', () => {
    opts.onPrimary();
    bubble.remove();
  });
  bubble.querySelector('.preview-secondary')?.addEventListener('click', () => {
    opts.onSecondary();
    bubble.remove();
  });
  container.appendChild(bubble);
  bubble.scrollIntoView({ behavior: 'smooth', block: 'end' });
  return bubble;
}
```

```ts
export function renderInsights(container: HTMLElement): void {
  const user = getCachedUser();
  const insights = (user?.insights ?? []).slice().sort((a, b) =>
    b.createdAt.localeCompare(a.createdAt));

  if (insights.length === 0) {
    // eslint-disable-next-line no-restricted-syntax -- trusted static template
    container.innerHTML = `
      <div class="insights-section" id="insightsTab">
        <h2 style="margin-bottom:16px;">💡 인사이트</h2>
        <div id="insightContent" style="color:var(--text-secondary);font-size:0.9rem;line-height:1.6;">아직 인사이트가 없어요. 대화를 나누고 인사이트 카드를 만들어보세요.</div>
      </div>
    `;
    return;
  }

  // eslint-disable-next-line no-restricted-syntax -- escapeHtml applied
  container.innerHTML = `
    <div class="insights-section" id="insightsTab">
      <h2 style="margin-bottom:16px;">💡 인사이트</h2>
      <div class="insights-grid">
        ${insights.map(i => `
          <button type="button" class="insight-card" data-insight-id="${escapeHtml(i.id)}">
            <span class="insight-text">${escapeHtml(i.text)}</span>
            <span class="insight-date">${escapeHtml(formatRelative(i.createdAt))}</span>
          </button>
        `).join('')}
      </div>
    </div>
  `;

  container.querySelectorAll<HTMLButtonElement>('.insight-card').forEach(btn => {
    btn.addEventListener('click', () => {
      const id = btn.dataset.insightId!;
      openInsightDetailModal(id);
    });
  });
}

export function mountInsightsHandlers(): void {
  on('dg:nav:tab-changed', ({ tab }) => {
    if (tab === 'insights') {
      const c = document.getElementById('insightsTab')?.parentElement;
      if (c) renderInsights(c as HTMLElement);
    }
  });
  on('dg:insights:added', () => {
    const c = document.getElementById('insightsTab')?.parentElement;
    if (c) renderInsights(c as HTMLElement);
  });
  on('dg:insights:removed', () => {
    const c = document.getElementById('insightsTab')?.parentElement;
    if (c) renderInsights(c as HTMLElement);
  });
}
```

```ts
// src/ui/modals/insight-detail.ts
import { openModal, closeModal } from './shared';
import { getCachedUser, saveUser } from '../../state/user';
import { escapeHtml } from '../../utils/escapeHtml';
import { showToast } from '../../utils/toast';
import { dispatch } from '../events';

export function openInsightDetailModal(id: string): void {
  const user = getCachedUser();
  const insight = user?.insights.find(i => i.id === id);
  if (!user || !insight) return;

  // eslint-disable-next-line no-restricted-syntax -- escapeHtml applied
  const body = `
    <div class="insight-detail">
      <p class="insight-detail-text">${escapeHtml(insight.text)}</p>
      <p class="insight-detail-date">${escapeHtml(insight.createdAt.slice(0, 10))}</p>
      <button type="button" class="btn btn-danger insight-delete-btn">삭제</button>
    </div>
  `;
  // T0 발견: ModalConfig field는 `bodyHtml` (not `content`) — shared.ts:6
  const modal = openModal({ title: '인사이트', bodyHtml: body });
  modal.querySelector('.insight-delete-btn')?.addEventListener('click', () => {
    if (!window.confirm('이 인사이트를 삭제할까요?')) return;
    user.insights = user.insights.filter(i => i.id !== id);
    saveUser(user);
    dispatch('dg:insights:removed', { id });
    closeModal();
    showToast('인사이트를 삭제했어요');
  });
}
```
