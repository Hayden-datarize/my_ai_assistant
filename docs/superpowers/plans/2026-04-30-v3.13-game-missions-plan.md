# DG v3.13 — Game Missions Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** v3.12 즉시 보상 + 18 뱃지 위에 **주기적 자극 레이어** 도입 — Daily 3 (풀 7) + Weekly 2 + Monthly 1 미션 시스템 + 미션 뱃지 4개 (Mission 카테고리). 캘린더 기준 KST 리셋, lazy regeneration, deterministic random (Daily만 풀에서 매일 3개).

**Architecture:** v3.12 sweep 패턴 그대로 재사용. `recordDailyAnswer` / `toggleScrap` / `saveMemo` 3 진입점에 `getActiveMissions(now, user) → prev=takeSnapshot → tickMissionProgress(user, action) → mutate → saveUser → curr=takeSnapshot → runSweep` 체인. `getActiveMissions`는 user mutate만 하고 `saveUser`는 caller 책임 (invariant). `runSweep`은 read-only — `detectEvents`에 `mission-complete` event detector 추가.

**Tech Stack:** TypeScript (strict) · Vite · Vitest jsdom · Playwright smoke · Vanilla DOM · `Intl.DateTimeFormat` for KST · mulberry32 PRNG (자작 ~10 LOC).

**Spec:** `docs/superpowers/specs/2026-04-30-v3.13-game-missions-design.md` (Status: Approved)

```ts
// src/state/migration.ts:80- (append)
import type { User } from './user';
import type { MissionInstance } from './missionTypes';   // T2에서 작성

/**
 * v3.13: schema v2 → v3 — missions 필드 추가.
 * - active 빈 배열, cumulative 0, ISO 필드 빈 문자열.
 * - 다음 getActiveMissions 호출 시 모든 주기 새로 생성.
 * - shape guard: missions.active 배열 아니면 [] 보강 (silent corruption 방지).
 */
export function migrateUserToV3(raw: unknown): User {
  const v2 = (raw as User & { schemaVersion?: number });
  if (v2?.schemaVersion === 3 && v2.missions && Array.isArray(v2.missions.active)) {
    // shape guard: 이미 v3이라도 cumulative number/string 필드 NaN 차단
    const m = v2.missions;
    m.cumulative = m.cumulative ?? { dailyCount: 0, weeklyCount: 0, monthlyCount: 0 };
    if (typeof m.cumulative.dailyCount !== 'number') m.cumulative.dailyCount = 0;
    if (typeof m.cumulative.weeklyCount !== 'number') m.cumulative.weeklyCount = 0;
    if (typeof m.cumulative.monthlyCount !== 'number') m.cumulative.monthlyCount = 0;
    if (typeof m.lastDailySeed !== 'string') m.lastDailySeed = '';
    if (typeof m.currentWeekIso !== 'string') m.currentWeekIso = '';
    if (typeof m.currentMonthIso !== 'string') m.currentMonthIso = '';
    return v2 as User;
  }
  // v1/v2 → v3
  return {
    ...(v2 as User),
    schemaVersion: 3,
    missions: {
      active: [],
      cumulative: { dailyCount: 0, weeklyCount: 0, monthlyCount: 0 },
      lastDailySeed: '',
      currentWeekIso: '',
      currentMonthIso: '',
    },
  };
}
```

```ts
import { migrateUserToV2, migrateUserToV3 } from './migration';

export function getCachedUser(): User | null {
  if (cached) return cached;
  try {
    const raw = localStorage.getItem(KEY);
    if (!raw) return null;
    const parsed = JSON.parse(raw);
    let user = (parsed?.schemaVersion === 2 || parsed?.schemaVersion === 3)
      ? parsed
      : migrateUserToV2(parsed);
    user = migrateUserToV3(user);                  // v2 → v3 (idempotent if already v3)
    if (!isValidUserShape(user)) return null;
    cached = user;
    return cached;
  } catch {
    return null;
  }
}
```

```ts
// src/state/missionEngine.ts
import type { User } from './user';
import type { MissionInstance, MissionPeriod } from './missionTypes';
import { DAILY_POOL, WEEKLY_FIXED, MONTHLY_FIXED } from './missionCatalog';

const KST_FMT = new Intl.DateTimeFormat('en-CA', { timeZone: 'Asia/Seoul', year: 'numeric', month: '2-digit', day: '2-digit' });

export function getKSTDateIso(now: Date): string {
  return KST_FMT.format(now);                                            // 'YYYY-MM-DD'
}

export function getKSTMonthIso(now: Date): string {
  return getKSTDateIso(now).slice(0, 7);                                 // 'YYYY-MM'
}

export function getKSTWeekIso(now: Date): string {
  // ISO 8601 week, KST 기준 (월요일 시작). 자체 계산 (Intl은 weekYear 미지원).
  const ymd = getKSTDateIso(now).split('-').map(Number);
  const utc = new Date(Date.UTC(ymd[0], ymd[1] - 1, ymd[2]));
  const day = utc.getUTCDay() || 7;                                      // Sun=0 → 7
  utc.setUTCDate(utc.getUTCDate() + 4 - day);                            // 같은 주 목요일
  const yearStart = new Date(Date.UTC(utc.getUTCFullYear(), 0, 1));
  const weekNum = Math.ceil(((utc.getTime() - yearStart.getTime()) / 86400000 + 1) / 7);
  return `${utc.getUTCFullYear()}-W${String(weekNum).padStart(2, '0')}`;
}

export function getKSTWindowStart(now: Date, period: MissionPeriod): number {
  const ymd = getKSTDateIso(now).split('-').map(Number);
  if (period === 'daily') {
    return Date.UTC(ymd[0], ymd[1] - 1, ymd[2]) - 9 * 3600 * 1000;       // KST 00:00 = UTC -9h
  }
  if (period === 'monthly') {
    return Date.UTC(ymd[0], ymd[1] - 1, 1) - 9 * 3600 * 1000;
  }
  // weekly: KST 월요일 00:00
  const utc = new Date(Date.UTC(ymd[0], ymd[1] - 1, ymd[2]));
  const day = utc.getUTCDay() || 7;
  utc.setUTCDate(utc.getUTCDate() - (day - 1));                          // 그 주 월요일
  return utc.getTime() - 9 * 3600 * 1000;
}

function hashStr(s: string): number {
  let h = 2166136261;
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return h >>> 0;
}

function mulberry32(seed: number): () => number {
  let a = seed;
  return () => {
    a |= 0; a = (a + 0x6d2b79f5) | 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export function pickDailyMissions(dateIso: string, now: Date): MissionInstance[] {
  const rng = mulberry32(hashStr(dateIso));
  const pool = [...DAILY_POOL];
  const picked: MissionInstance[] = [];
  for (let i = 0; i < 3 && pool.length; i++) {
    const idx = Math.floor(rng() * pool.length);
    const def = pool.splice(idx, 1)[0];
    const inst: MissionInstance = {
      defId: def.id, period: 'daily',
      windowStart: getKSTWindowStart(now, 'daily'),
      progress: 0, completed: false,
    };
    if (def.triggerOn === 'active-day') inst.progressDates = [];
    picked.push(inst);
  }
  return picked;
}

function makeFixed(defs: readonly { id: string; period: MissionPeriod; triggerOn: string }[], period: MissionPeriod, now: Date): MissionInstance[] {
  return defs.map(def => {
    const inst: MissionInstance = {
      defId: def.id, period,
      windowStart: getKSTWindowStart(now, period),
      progress: 0, completed: false,
    };
    if (def.triggerOn === 'active-day') inst.progressDates = [];
    return inst;
  });
}

/**
 * Lazy regeneration. expired 미션은 새로 생성, 미완수 carry-over X.
 * **caller invariant**: saveUser는 caller 책임 (이중 saveUser race 회피).
 */
export function getActiveMissions(now: Date, user: User): MissionInstance[] {
  const todayIso = getKSTDateIso(now);
  const weekIso = getKSTWeekIso(now);
  const monthIso = getKSTMonthIso(now);

  if (user.missions.lastDailySeed !== todayIso) {
    const newDaily = pickDailyMissions(todayIso, now);
    user.missions.active = user.missions.active.filter(m => m.period !== 'daily').concat(newDaily);
    user.missions.lastDailySeed = todayIso;
  }
  if (user.missions.currentWeekIso !== weekIso) {
    const newWeekly = makeFixed(WEEKLY_FIXED, 'weekly', now);
    user.missions.active = user.missions.active.filter(m => m.period !== 'weekly').concat(newWeekly);
    user.missions.currentWeekIso = weekIso;
  }
  if (user.missions.currentMonthIso !== monthIso) {
    const newMonthly = makeFixed(MONTHLY_FIXED, 'monthly', now);
    user.missions.active = user.missions.active.filter(m => m.period !== 'monthly').concat(newMonthly);
    user.missions.currentMonthIso = monthIso;
  }
  return user.missions.active;
}
```

```ts
// src/state/missionEngine.ts (append)
import { getMissionDef } from './missionCatalog';
import type { MissionAction } from './missionTypes';

/**
 * progress++ → completed 전환 시 xp 보너스 + cumulative 카운트.
 * **caller invariant**: tick 후 caller가 saveUser 호출 (race 회피, sweep은 read-only).
 * active-day trigger는 progressDates 기반 idempotent (같은 날 중복 push 차단).
 */
export function tickMissionProgress(user: User, action: MissionAction, now: Date = new Date()): void {
  const todayIso = getKSTDateIso(now);
  for (const m of user.missions.active) {
    if (m.completed) continue;
    const def = getMissionDef(m.defId);
    if (!def) continue;                              // catalog 변경 시 unknown defId 방어

    if (def.triggerOn === 'active-day') {
      if (action === 'active-day') continue;         // 'active-day' raw action은 호출되지 않음 (방어)
      m.progressDates = m.progressDates ?? [];
      if (m.progressDates.includes(todayIso)) continue;
      m.progressDates.push(todayIso);
      m.progress = m.progressDates.length;
    } else if (def.triggerOn === action) {
      m.progress++;
    } else {
      continue;
    }

    if (m.progress >= def.target && !m.completed) {
      m.completed = true;
      user.xp += def.rewardXp;
      const key = `${m.period}Count` as 'dailyCount' | 'weeklyCount' | 'monthlyCount';
      user.missions.cumulative[key]++;
    }
  }
}
```

```ts
  monthly: '이번 달 도전',

const PERIODS: MissionPeriod[] = ['daily', 'weekly', 'monthly'];

function escapeHtml(s: string): string {
  return s.replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]!));
}

function renderCard(m: MissionInstance): string {
  const def = getMissionDef(m.defId);
  if (!def) return '';
  const cls = m.completed ? 'mission-card mission-card--completed' : 'mission-card';
  const pbHtml = def.target >= 2
    ? `<span class="mission-card__progress" role="progressbar" aria-valuenow="${m.progress}" aria-valuemax="${def.target}" aria-label="${m.progress}/${def.target}">
         <span class="mission-card__bar" style="width:${Math.min(100, (m.progress / def.target) * 100)}%"></span>
         <span class="mission-card__count">${m.progress}/${def.target}</span>
       </span>`
    : '';
  const checkHtml = m.completed ? '<span class="mission-card__check" aria-label="완수">✓</span>' : '';
  return `<li class="${cls}">
    <span class="mission-card__title">${escapeHtml(def.text)}</span>
    ${pbHtml}
    <span class="mission-card__reward">+${def.rewardXp} XP</span>
    ${checkHtml}
  </li>`;
}

function renderGroup(period: MissionPeriod, missions: MissionInstance[]): string {
  if (!missions.length) return '';
  const completedCount = missions.filter(m => m.completed).length;
  const listId = `missionGroup${period.charAt(0).toUpperCase() + period.slice(1)}`;
  return `<div class="mission-group" data-period="${period}">
    <button class="mission-group__header" type="button" aria-expanded="true" aria-controls="${listId}">
      <span class="mission-group__title">${GROUP_TITLE[period]}</span>
      <span class="mission-group__progress">${completedCount}/${missions.length}</span>
      <span class="mission-group__chevron" aria-hidden="true">▼</span>
    </button>
    <ul id="${listId}" class="mission-group__list" role="list">
      ${missions.map(renderCard).join('')}
    </ul>
  </div>`;
}
```

```ts
import type { MissionAction } from '../../state/missionTypes';

function fireTrigger(action: MissionAction): void {
  const u = getCachedUser();
  if (!u) return;
  getActiveMissions(new Date(), u);
  const prev = takeSnapshot();
  tickMissionProgress(u, action);
  saveUser(u);
  const curr = takeSnapshot();
  runSweep(prev, curr);
}

export function fireBriefingViewTrigger(): void { fireTrigger('briefing-view'); }
export function fireArchiveRevisitTrigger(): void { fireTrigger('archive-revisit'); }
export function fireCrossInterestTrigger(): void { fireTrigger('cross-interest-view'); }
```

```ts
describe('edge cases', () => {
```
