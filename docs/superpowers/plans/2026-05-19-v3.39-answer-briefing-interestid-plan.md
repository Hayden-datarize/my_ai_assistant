// src/state/migration.ts (파일 끝 append)

/**
 * v3.39 T1: schema v8 → v9 — Answer/Briefing interestId 도입.
 * Answer/Briefing은 별도 storage (loadAnswers/loadBriefings boundary에서 normalize) → User 객체 자체엔 ripple 0.
 * chain superset (v3.25 T2 패턴): V2~V8 모든 early-return이 v9 user에 same-reference pass-through.
 *
 * @internal Caller invariant — `getCachedUser` chain (V2→V3→V4→V5→V6→V7→V8→V9) 후 호출.
 */
export function migrateUserToV9(u: unknown): User {
  const r = u as Record<string, unknown> & { schemaVersion?: number };

  if (r.schemaVersion === 9) {
    return r as unknown as User; // idempotent
  }

  // v8까지 lift (V8 early-return이 v9 guard됨 — 불필요 이중 lift 없음)
  const v8 = migrateUserToV8(u);
  const v8r = v8 as unknown as Record<string, unknown>;
  return { ...v8r, schemaVersion: 9 } as unknown as User;
}
```

// src/utils/interestKeywords.ts 끝에 추가
import { validateInterestId } from '../state/user';

/**
 * v3.39 T2 (Codex 사전 P0-3): text가 interestId 분야에 속하는지 판정.
 *
 * spec v1의 `matchKeyword(text, interestId)` 직접 호출은 false-negative 함정:
 * - matchKeyword은 keyword token (e.g., '채용')용이지 interest id (e.g., 'ai_ml')용 아님.
 *
 * 본 helper는:
 * 1. id를 validateInterestId로 통과 검사 — invalid 즉시 false.
 * 2. interestKeywords(id) 토큰 목록 중 하나라도 text에 match → true.
 * 3. lowercase 정규화 (text + keyword 모두).
 *
 * @internal Caller: chat-summary inferredInterestId / Briefing storage / archive legacy fallback.
 */
export function matchesInterest(text: string, id: string): boolean {
  if (validateInterestId(id) === 'unknown') return false;
  if (!text) return false;
  const keywords = interestKeywords(id);
  const hay = text.toLowerCase();
  return keywords.some((kw) => matchKeyword(hay, kw.toLowerCase()));
}
```

/**
 * v3.39 T3 (Codex 사전 P1-2): generateQuestion JSON 응답 parse + interestId validate fallback.
 *
 * 4-step 폴백:
 * 1. JSON parse fail → throw (caller가 fallbackQuestion으로 catch)
 * 2. interestId 누락 또는 invalid → userInterests[0] 폴백
 * 3. userInterests[0]도 없으면 'unknown'
 *
 * @internal Caller: home.ts submitAnswer flow (live + cached path 둘 다 통일).
 */
export function parseQuestionResponse(raw: string, userInterests: string[]): QuestionResponse {
  const data = parseJsonText<{ type?: string; question?: string; hint?: string; interestId?: string }>(raw);
  const type = typeof data.type === 'string' ? data.type : '분석';
  const question = typeof data.question === 'string' ? data.question : '';
  const hint = typeof data.hint === 'string' ? data.hint : '';
  const rawId = typeof data.interestId === 'string' ? data.interestId : '';
  const validated = rawId ? validateInterestId(rawId) : 'unknown';
  const targetInterestId = validated === 'unknown' ? (userInterests[0] ?? 'unknown') : validated;
  return { type, question, hint, targetInterestId };
}

export function resolveQuestionInterestId(rawId: string | undefined, userInterests: string[]): string {
  if (!rawId) return userInterests[0] ?? 'unknown';
  const validated = validateInterestId(rawId);
  return validated === 'unknown' ? (userInterests[0] ?? 'unknown') : validated;

type AnyArchiveEntity = Answer | Briefing | Insight;

/** v3.39 T6 (Codex P0-2): archive 분야 filter 단일 predicate.
 * 1순위 exact / 2순위 legacy unknown + text matchesInterest fallback.
 */
export function entityMatchesInterest(e: AnyArchiveEntity, id: string | null): boolean {
  if (id === null) return true;
  if (!('interestId' in e)) return false;
  if (e.interestId === id) return true;
  if (e.interestId === 'unknown') {
    const text = getEntityText(e);
    return matchesInterest(text, id);
  }
  return false;
}

function getEntityText(e: AnyArchiveEntity): string {
  // Answer: text + questionText / Briefing: title + summary + memo / Insight: text
  if ('text' in e && 'questionId' in e) return `${e.text} ${e.questionText ?? ''}`;
  if ('title' in e && 'summary' in e) return `${e.title} ${e.summary} ${e.memo}`;
  if ('text' in e) return e.text;
  return '';
}
