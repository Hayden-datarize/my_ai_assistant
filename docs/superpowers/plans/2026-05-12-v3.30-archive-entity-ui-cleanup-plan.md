export function renderEntityChipRow(
  active: EntityFilter = 'all',
  counts: Record<EntityFilter, number> | null = null,
): string {
  const chips = ENTITY_ORDER.map((id) => {
    const checked = id === active ? 'true' : 'false';
    const activeClass = id === active ? ' active' : '';
    const countSpan = counts !== null
      ? `<span class="archive-entity-count" data-entity-count="${id}">${counts[id]}</span>`
      : '';
    return `<button type="button" role="radio" class="archive-entity-chip${activeClass}" data-entity="${id}" aria-checked="${checked}">${ENTITY_LABELS[id]}${countSpan}</button>`;
  }).join('');
  return `<div id="archiveEntityFilters" role="radiogroup" aria-label="아카이브 카테고리" class="archive-entity-row">${chips}</div>`;
}
```

import { getAllAnswers } from '../../state/answers';
import { getScraps } from '../../state/scraps';
import { getInsights } from '../../state/insights';

/** v3.30 T2: entity별 count 단일 source-of-truth helper. */
export function getEntityCounts(): Record<EntityFilter, number> {
  const answer = getAllAnswers().length;
  const scrap = getScraps().length;
  const insight = getInsights().length;
  return { all: answer + scrap + insight, answer, scrap, insight };
}

/** v3.30 T2: DOM data-entity-count span 4개 textContent 갱신 (chip row 재생성 X). */
export function refreshEntityCounts(counts?: Record<EntityFilter, number>): void {
  const c = counts ?? getEntityCounts();
  document.querySelectorAll<HTMLElement>('[data-entity-count]').forEach((el) => {
    const id = el.getAttribute('data-entity-count') as EntityFilter | null;
    if (id && c[id] !== undefined) el.textContent = String(c[id]);
  });
}

// mountArchiveHandlers 안 listener:
document.addEventListener('dg:archive:updated', () => {
  refreshEntityCounts();
});
```
