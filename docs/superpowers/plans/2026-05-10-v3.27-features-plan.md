      }));

      return {
        ...v7,
        schemaVersion: 8 as const,
        insights,
      } as unknown as User;
    }

  import { LEVEL_THRESHOLDS } from '../../state/levels';  // 또는 실 위치

  interface XpHistoryEntry { date: string; cumulativeXp: number; }

  export function renderXpChart(history: XpHistoryEntry[]): SVGElement {
    const svg = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
    svg.setAttribute('viewBox', '0 0 300 80');
    svg.setAttribute('aria-label', buildAriaLabel(history));
    if (history.length < 2) {
      svg.appendChild(emptyStateText('아직 데이터가 없어요'));
      return svg;
    }
    const path = buildPath(history);  // 'M x0 y0 L x1 y1 ...'
    svg.appendChild(svgPath(path));
    history.forEach((h, idx) => {
      // levelup 감지: prev cumulative와 cross threshold
      if (idx > 0 && crossLevelThreshold(history[idx-1].cumulativeXp, h.cumulativeXp)) {
        svg.appendChild(svgMarker(idx, h));
      }
    });
    return svg;
  }
  function buildAriaLabel(h: XpHistoryEntry[]): string {
    if (h.length === 0) return '최근 30일 XP 추이: 데이터 없음';
    const start = h[0].cumulativeXp, end = h[h.length-1].cumulativeXp;
    const max = Math.max(...h.map(e => e.cumulativeXp));
    const levelups = countLevelups(h);
    return `최근 30일 XP 추이: 시작 ${start}, 끝 ${end}, 최대 ${max}, 레벨업 ${levelups}회`;
  }
