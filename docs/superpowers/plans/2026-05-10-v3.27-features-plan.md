  ```ts
  export function migrateUserToV8(u: unknown): User {
    const r = u as Record<string, unknown> & { schemaVersion?: number };
    if (r.schemaVersion === 8) return r as unknown as User;

    const v7 = migrateUserToV7(u);
    const v7r = v7 as unknown as Record<string, unknown>;

      pinned: typeof i.pinned === 'boolean' ? i.pinned : false,

      ...v7,
      schemaVersion: 8 as const,
      xpHistory: Array.isArray(v7r.xpHistory) ? v7r.xpHistory : [],

    } as unknown as User;
  }
  ```

  ```ts
  // src/state/user.ts 끝부분
  export interface XpHistoryEntry { date: string; cumulativeXp: number; }

  /** 최근 N일 cumulative XP — 없는 날은 직전 cumulative 유지 */
  export function getXpHistory(days: number = 30): XpHistoryEntry[] {
    const user = getCachedUser();
    if (!user) return [];
    const raw = user.xpHistory.slice(-days * 2);  // 여유분 (같은 날 다중 push 가능)
    if (raw.length === 0) return [];
    // KST 일별 그룹화 → cumulative 변환
    const byDate = new Map<string, number>();
    for (const h of raw) {
      byDate.set(h.date, (byDate.get(h.date) ?? 0) + h.xpEarned);
    }
    let cumulative = 0;
    const result: XpHistoryEntry[] = [];
    for (const [date, earned] of [...byDate.entries()].sort(([a], [b]) => a.localeCompare(b))) {
      cumulative += earned;
      result.push({ date, cumulativeXp: cumulative });
    }
    return result.slice(-days);
  }
  ```
